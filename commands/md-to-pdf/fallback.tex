% Switched as runs to DejaVu Sans (defining them one by one, like the

\count255="0370

\XeTeXinterchartoks \greekclass 4095 = {\endgroup}

% --- Hindi / Devanagari ---
% Hindi can't be routed letter-by-letter (the vowel signs and conjuncts
% must be shaped together), so a whole run of Devanagari text switches
% to the first Devanagari font found on this computer. If none is
% installed the text still prints nothing -- install one, e.g.
%   Ubuntu/Debian:  sudo apt install fonts-noto-core   (or fonts-lohit-deva)
%   macOS:          already has 'Kohinoor Devanagari' / 'Devanagari Sangam MN'
%   Windows:        already has 'Nirmala UI' / 'Mangal'
\def\devafontname{}
\def\trydevafont#1{\ifx\devafontname\empty\IfFontExistsTF{#1}{\def\devafontname{#1}}{}\fi}
\trydevafont{Noto Sans Devanagari}\trydevafont{Noto Serif Devanagari}
\trydevafont{Lohit Devanagari}\trydevafont{Kohinoor Devanagari}
\trydevafont{Devanagari Sangam MN}\trydevafont{ITF Devanagari}
\trydevafont{Nirmala UI}\trydevafont{Mangal}\trydevafont{Gargi}
\ifx\devafontname\empty\else
  \newfontfamily\devafont{\devafontname}[Script=Devanagari]
  \AtBeginDocument{\XeTeXinterchartokenstate=1}
  \newXeTeXintercharclass\devaclass
  \count255="0900
  \loop\XeTeXcharclass\count255=\devaclass\ifnum\count255<"097F\advance\count255 by 1\repeat
  \XeTeXcharclass"200C=\devaclass \XeTeXcharclass"200D=\devaclass
  \XeTeXinterchartoks 0 \devaclass = {\begingroup\devafont}
  \XeTeXinterchartoks 4095 \devaclass = {\begingroup\devafont}
  \XeTeXinterchartoks \devaclass 0 = {\endgroup}
  \XeTeXinterchartoks \devaclass 4095 = {\endgroup}
\fi
